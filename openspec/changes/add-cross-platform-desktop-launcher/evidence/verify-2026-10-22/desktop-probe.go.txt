func TestVerifyAtomicWriteFailure(t *testing.T) {
 if os.Getenv("PIWORK_VERIFY_FSIZE_CHILD")=="1" {
  dir:=privateTemp(t);target:=filepath.Join(dir,"current.log")
  var old syscall.Rlimit;if err:=syscall.Getrlimit(syscall.RLIMIT_FSIZE,&old);err!=nil{t.Fatal(err)}
  signal.Ignore(syscall.SIGXFSZ);limit:=old;limit.Cur=1024
  if err:=syscall.Setrlimit(syscall.RLIMIT_FSIZE,&limit);err!=nil{t.Fatal(err)}
  err:=writeAtomic(target,bytes.Repeat([]byte("x"),16384),0600)
  syscall.Setrlimit(syscall.RLIMIT_FSIZE,&old)
  st,statErr:=os.Stat(target)
  if err==nil {if statErr==nil{t.Fatalf("write error was lost: requested=16384 actual=%d, reported success",st.Size())};t.Fatal("write error was lost")};return
 }
 cmd:=exec.Command(os.Args[0],"-test.run=^TestVerifyAtomicWriteFailure$","-test.v");cmd.Env=append(os.Environ(),"PIWORK_VERIFY_FSIZE_CHILD=1")
 output,err:=cmd.CombinedOutput();if err!=nil{t.Fatalf("isolated size-limited writer:\n%s",output)}
}
func TestVerifyFinishStopRace(t *testing.T) {
 c:=testController(t,&fakeAdapter{},nil,privateTemp(t))
 for i:=0;i<2000;i++ {
  s:=&session{id:"probe",cancel:func(){},done:make(chan struct{})};c.mu.Lock();c.session=s;c.mu.Unlock()
  start:=make(chan struct{});var wg sync.WaitGroup;wg.Add(2)
  go func(){defer wg.Done();<-start;c.Stop()}()
  go func(){defer wg.Done();<-start;c.finish(s,errors.New("Guest exited unexpectedly"))}()
  close(start);wg.Wait()
 }
}
